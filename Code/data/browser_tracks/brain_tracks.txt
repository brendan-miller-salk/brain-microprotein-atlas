# smORF peptide evidence: PROSIT-graded tryptic peptides projected to the genome.
# bigGenePred, so UCSC translates the CDS and draws the amino-acid letters the
# same way it does for GENCODE, while itemRgb carries the PROSIT confidence tier.
# The amino-acid letters come from baseColorDefault=genomicCodons on the track line.
# They are a DISPLAY setting, not something in the file - this is documented for custom
# tracks in UCSC's own bigGenePred example ("Display Amino Acids and Codon Numbers").
# Equivalently, right-click the track -> Configure -> "Color track by codons" ->
# "genomic codons", and tick "Show codon numbering".
# green=Strong (1,177)  gold=Moderate (1,710)  firebrick=Weak (417)
# grey=Insufficient/ungraded (550).  3,854 peptides, every one verified to
# translate to its expected residues from hg38.  Built by transcript_rules/make_psm_biggenepred.sh (Box).

track type=bigGenePred baseColorDefault=genomicCodons name="smORF PROSIT Peptides" description="PROSIT-graded peptides on smORFs, colored by confidence (green=Strong gold=Moderate firebrick=Weak grey=Insufficient)" visibility=pack itemRgb=on priority=5 bigDataUrl=https://huggingface.co/datasets/brmiller/brain-microprotein-atlas/resolve/main/browser_tracks/smorf_psm_peptides.bb

# smORF transcript-architecture ORF rules: each smORF colored by its predicted
# decay fate from nmd_analysis/ (Box) crossed with whether the host gene's main
# ORF is disrupted in the transcripts that carry it.
# bigGenePred with itemRgb: the CDS frames let UCSC draw the amino acids, exactly
# as on the peptide track. It was previously plain BED12/type=bigBed, which has no
# frame column - so no track-line setting could make it show residues.
#
# Two decay rules, scored in this order:
#   NSD (non-stop decay) - the ORF runs off the transcript with no in-frame stop.
#     Unlike NMD this CANNOT be judged from a single transcript, so it is decided
#     per smORF over the whole compatible set: at least one confident no-stop
#     transcript AND no transcript carrying a real stop codon. 21 smORFs have a
#     confident no-stop transcript but are ruled out by that contradiction and
#     stay green/gold/firebrick; nsd_gate_reason in the assignments TSV records
#     which. NSD overrides the NMD call, so it is a third decay state, not a
#     fifth colour.
#   NMD - the stop codon trips the 50-nt rule on a representative transcript.
#
# NOTE the green/gold/firebrick trio is shared with the PROSIT peptide track
# above, where those colours mean something else entirely (firebrick = "Weak
# PROSIT match" there, "NMD, ORF intact" here). Kept deliberately - it is the
# published palette. The two purples are deliberately OUTSIDE that shared set,
# so an NSD call is never ambiguous when both tracks are loaded together. UCSC
# has no per-track background colour (no bgColor setting), so if the two tracks
# ever need separating further, the way to do it is to move one to a different
# hue family.
# dark green   0,100,0     no NMD, main ORF disrupted   (3,096)
# light green  144,238,144 no NMD, main ORF intact      (1,044)
# gold         255,215,0   NMD, main ORF disrupted        (771)
# firebrick    178,34,34   NMD, main ORF intact           (444)
# dark purple  106,61,154  NSD, main ORF disrupted        (183)
# light purple 188,158,217 NSD, main ORF intact             (83)
# grey         128,128,128 no data                         (47)
# Source BED is alongside as smorf_trx_priority_color_class.bed; the .bb is what
# UCSC loads. Built by transcript_rules/make_orf_rules_biggenepred.sh, which verifies
# every CDS frame against the genome (120 TrEMBL records needed correcting; all
# 5,621 coding features now translate to a clean ORF with no internal stop).
# Those 120 are itemised in trembl_cds_rephase.tsv alongside - every one had at
# least one internal stop in its annotated frame and none ended in a stop codon;
# 51 were shifted by 1 nt and 69 by 2.
# Each thick (CDS) region now runs through its stop codon, so UCSC draws the
# terminal * residue: 4,933 records were extended, either onto an annotated
# GENCODE stop_codon feature or onto the next genomic triplet walked along the
# parent exon chain (the two routes agree wherever both apply). Records whose
# model is 3'-incomplete are left as they were - a stop is not invented.

track type=bigGenePred baseColorDefault=genomicCodons name="smORF Rules" description="smORFs colored by decay rule and main-ORF status (dark green=no NMD/ORF disrupted, light green=no NMD/ORF intact, gold=NMD/ORF disrupted, firebrick=NMD/ORF intact, dark purple=NSD/ORF disrupted, light purple=NSD/ORF intact, grey=no data)" visibility=pack itemRgb=on priority=6 bigDataUrl=https://huggingface.co/datasets/brmiller/brain-microprotein-atlas/resolve/main/browser_tracks/smorf_trx_priority_color_class.bb
